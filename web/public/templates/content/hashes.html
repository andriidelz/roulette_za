package main

import (
    "crypto/sha256"
    "encoding/hex"
    "fmt"
)

func main() {
    number := 5                   // Number that came up
    salt := "a1b2c3d4e5f6"        // Salt from bot message
    originalHash := "e37f...0840" // Hash from round start

    // Format string and calculate hash
    data := fmt.Sprintf("%d:%s", number, salt)
    hasher := sha256.New()
    hasher.Write([]byte(data))
    computedHash := hex.EncodeToString(hasher.Sum(nil))

    // Verify match
    if computedHash == originalHash {
        fmt.Println("Result is valid!")
    } else {
        fmt.Println("Result does not match hash!")
    }
}

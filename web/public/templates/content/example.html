// Function for SHA-256 hash verification
async function verifyHash(number, salt, originalHash) {
    try {
        // Format string as "number:salt"
        const data = `${number}:${salt}`;

        // Encode string to byte array
        const encoder = new TextEncoder();
        const dataBuffer = encoder.encode(data);

        // Calculate SHA-256 hash
        const hashBuffer = await crypto.subtle.digest('SHA-256', dataBuffer);

        // Convert ArrayBuffer to hex string
        const hashArray = Array.from(new Uint8Array(hashBuffer));
        const computedHash = hashArray.map(b =>
            b.toString(16).padStart(2, '0')
        ).join('');

        // Check if computed hash matches the original
        const isValid = computedHash === originalHash;

        return {
            isValid,
            computedHash,
            originalHash,
            number,
            salt,
            input: data
        };
    } catch (error) {
        console.error('Error during hash verification:', error);
        return {
            isValid: false,
            error: error.message
        };
    }
}
